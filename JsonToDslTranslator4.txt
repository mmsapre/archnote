
import org.springframework.stereotype.Component;

import java.util.*;
import java.util.stream.Collectors;

@Component
@SuppressWarnings("unchecked")
public class JsonToDslTranslator {

    private final QueryValidator queryValidator;
    private final TableConfig tableConfig;

    public JsonToDslTranslator(
            QueryValidator queryValidator,
            TableConfig tableConfig) {

        this.queryValidator = queryValidator;
        this.tableConfig = tableConfig;
    }

    /*
     * =========================================================
     * ENTRY POINT
     * =========================================================
     *
     * Existing:
     *
     * {
     *   "payload": {
     *      "source": [...],
     *      "from": ...
     *   }
     * }
     *
     * New:
     *
     * {
     *   "payload": {
     *      "with": [...],
     *      "source": [...]
     *   }
     * }
     *
     * WITH is optional, therefore old payloads continue working.
     * =========================================================
     */

    public String translate(Map<String, Object> queryRequest) {

        Map<String, Object> normalizedRequest =
                normalizeQueryRequest(queryRequest);

        Object withObject =
                normalizedRequest.get("with");

        if (withObject == null) {
            return translateQuery(
                    normalizedRequest,
                    Collections.emptyList());
        }

        if (!(withObject instanceof List)) {
            throw new IllegalArgumentException(
                    "'with' must be a list");
        }

        List<Map<String, Object>> ctes =
                (List<Map<String, Object>>) withObject;

        if (ctes.isEmpty()) {

            Map<String, Object> copy =
                    new LinkedHashMap<>(normalizedRequest);

            copy.remove("with");

            return translateQuery(
                    copy,
                    Collections.emptyList());
        }

        return translateWithClause(
                normalizedRequest,
                ctes);
    }


    /*
     * =========================================================
     * TRANSLATE ONE QUERY
     * =========================================================
     */

    public String translateQuery(
            Map<String, Object> queryRequest,
            List<Map<String, Object>> ctes) {

        Map<String, Object> normalizedRequest =
                normalizeQueryRequest(queryRequest);

        /*
         * Keep your existing validator.
         *
         * NOTE:
         * QueryValidator should eventually understand that a
         * FROM name can also be a CTE.
         */
        if (!queryValidator.validateQuery(queryRequest)) {

            throw new IllegalArgumentException(
                    "Invalid table, alias, or columns in the request.");
        }

        /*
         * SELECT
         */
        List<String> selectColumns =
                extractSelectColumns(normalizedRequest);

        if (selectColumns.isEmpty()) {
            throw new IllegalArgumentException(
                    "No source fields found in request.");
        }

        String select =
                String.join(", ", selectColumns);

        /*
         * CTE names are needed so FROM base does not become
         * schema.base.
         */
        Set<String> cteNames =
                extractCteNames(ctes);

        /*
         * FROM + JOIN
         */
        String from =
                handleFromClause(
                        normalizedRequest,
                        cteNames);

        StringBuilder queryString =
                new StringBuilder();

        queryString
                .append("SELECT ")
                .append(select)
                .append(" FROM ")
                .append(from);

        /*
         * WHERE
         */
        if (normalizedRequest.containsKey("query")) {

            Object queryObject =
                    normalizedRequest.get("query");

            if (queryObject instanceof Map) {

                String where =
                        handleQueryClause(
                                (Map<String, Object>) queryObject);

                if (where != null &&
                        !where.isBlank()) {

                    queryString
                            .append(" WHERE ")
                            .append(where);
                }
            }
        }

        /*
         * GROUP BY
         */
        appendGroupBy(
                queryString,
                normalizedRequest);

        /*
         * HAVING
         */
        appendHaving(
                queryString,
                normalizedRequest);

        /*
         * ORDER BY
         */
        appendOrderBy(
                queryString,
                normalizedRequest);

        /*
         * Existing SIZE
         */
        if (normalizedRequest.containsKey("size")) {

            int limit =
                    parseLimit(
                            normalizedRequest.get("size"));

            queryString
                    .append(" LIMIT ")
                    .append(limit);

        } else if (
                normalizedRequest.get("pagination")
                        instanceof Map) {

            Map<String, Object> pagination =
                    (Map<String, Object>)
                            normalizedRequest.get("pagination");

            if (pagination.containsKey("limit")) {

                int limit =
                        parseLimit(
                                pagination.get("limit"));

                queryString
                        .append(" LIMIT ")
                        .append(limit);
            }
        }

        /*
         * Existing OFFSET
         */
        if (normalizedRequest.containsKey("offset")) {

            int offset =
                    parseOffset(
                            normalizedRequest.get("offset"));

            queryString
                    .append(" OFFSET ")
                    .append(offset);

        } else if (
                normalizedRequest.get("pagination")
                        instanceof Map) {

            Map<String, Object> pagination =
                    (Map<String, Object>)
                            normalizedRequest.get("pagination");

            if (pagination.containsKey("offset")) {

                int offset =
                        parseOffset(
                                pagination.get("offset"));

                queryString
                        .append(" OFFSET ")
                        .append(offset);
            }
        }

        System.out.println(
                "Query =========================");

        System.out.println(
                queryString);

        System.out.println(
                "End Query =====================");

        return queryString.toString();
    }


    /*
     * =========================================================
     * WITH
     * =========================================================
     */

    private String translateWithClause(
            Map<String, Object> normalizedRequest,
            List<Map<String, Object>> ctes) {

        if (ctes == null || ctes.isEmpty()) {

            Map<String, Object> query =
                    new LinkedHashMap<>(normalizedRequest);

            query.remove("with");

            return translateQuery(
                    query,
                    Collections.emptyList());
        }

        /*
         * Also validates duplicate CTE names.
         */
        extractCteNames(ctes);

        StringBuilder sql =
                new StringBuilder("WITH ");

        for (int i = 0;
             i < ctes.size();
             i++) {

            Map<String, Object> cte =
                    ctes.get(i);

            if (cte == null) {

                throw new IllegalArgumentException(
                        "CTE definition cannot be null");
            }

            String cteName =
                    getRequiredString(
                            cte,
                            "name");

            validateIdentifier(cteName);

            Object queryObject =
                    cte.get("query");

            if (!(queryObject instanceof Map)) {

                throw new IllegalArgumentException(
                        "CTE query is required for: "
                                + cteName);
            }

            Map<String, Object> cteQuery =
                    (Map<String, Object>) queryObject;

            String cteSql =
                    translateQuery(
                            cteQuery,
                            ctes);

            if (i > 0) {
                sql.append(", ");
            }

            sql
                    .append(cteName)
                    .append(" AS (")
                    .append(cteSql)
                    .append(")");
        }

        Map<String, Object> mainQuery =
                new LinkedHashMap<>(normalizedRequest);

        mainQuery.remove("with");

        String mainSql =
                translateQuery(
                        mainQuery,
                        ctes);

        sql
                .append(" ")
                .append(mainSql);

        return sql.toString();
    }


    /*
     * =========================================================
     * CTE NAMES
     * =========================================================
     */

    private Set<String> extractCteNames(
            List<Map<String, Object>> ctes) {

        Set<String> names =
                new HashSet<>();

        if (ctes == null) {
            return names;
        }

        for (Map<String, Object> cte : ctes) {

            if (cte == null) {
                continue;
            }

            String name =
                    getRequiredString(
                            cte,
                            "name");

            validateIdentifier(name);

            String normalizedName =
                    name.toLowerCase(
                            Locale.ROOT);

            if (!names.add(normalizedName)) {

                throw new IllegalArgumentException(
                        "Duplicate CTE: " + name);
            }
        }

        return names;
    }


    /*
     * =========================================================
     * NORMALIZE
     *
     * THIS MATCHES YOUR SCREENSHOT.
     *
     * Input:
     *
     * {
     *   "mrdrequestcontext": {...},
     *   "mrdrequestheader": {...},
     *   "payload": {
     *       ...
     *   }
     * }
     *
     * Translator works only on payload.
     * =========================================================
     */

    private Map<String, Object> normalizeQueryRequest(
            Map<String, Object> queryRequest) {

        if (queryRequest == null) {
            return Collections.emptyMap();
        }

        Object payload =
                queryRequest.get("payload");

        if (payload instanceof Map) {

            return (Map<String, Object>)
                    payload;
        }

        return queryRequest;
    }


    /*
     * =========================================================
     * SELECT
     *
     * EXISTING SCREENSHOT:
     *
     * "source": [
     *   {
     *      "field":"aiacashno.alternate_tp_value",
     *      "alias":"cashAccountNumber"
     *   }
     * ]
     *
     * NEW aggregate is just another source item.
     * =========================================================
     */

    private List<String> extractSelectColumns(
            Map<String, Object> queryRequest) {

        Object source =
                queryRequest.get("source");

        if (!(source instanceof List)) {
            return new ArrayList<>();
        }

        List<String> columns =
                new ArrayList<>();

        for (Object item :
                (List<?>) source) {

            if (!(item instanceof Map)) {
                continue;
            }

            Map<String, Object> selectItem =
                    (Map<String, Object>) item;

            /*
             * EXISTING FIELD
             */
            if (selectItem.containsKey("field")) {

                Object fieldObject =
                        selectItem.get("field");

                if (!(fieldObject instanceof String) ||
                        ((String) fieldObject).isBlank()) {

                    continue;
                }

                StringBuilder column =
                        new StringBuilder(
                                ((String) fieldObject).trim());

                Object aliasObject =
                        selectItem.get("alias");

                if (aliasObject instanceof String &&
                        !((String) aliasObject).isBlank()) {

                    column
                            .append(" AS ")
                            .append(
                                    formatAlias(
                                            (String) aliasObject));
                }

                columns.add(
                        column.toString());

                continue;
            }

            /*
             * NEW AGGREGATE
             */
            if (selectItem.containsKey("aggregate")) {

                columns.add(
                        buildAggregate(
                                selectItem));
            }
        }

        return columns;
    }


    /*
     * =========================================================
     * AGGREGATE
     * =========================================================
     *
     * {
     *   "aggregate": {
     *      "function":"MAX",
     *      "field":"b.instruct_value"
     *   },
     *   "alias":"gcsLeId"
     * }
     * =========================================================
     */

    private String buildAggregate(
            Map<String, Object> selectItem) {

        Object aggregateObject =
                selectItem.get("aggregate");

        if (!(aggregateObject instanceof Map)) {

            throw new IllegalArgumentException(
                    "aggregate must be an object");
        }

        Map<String, Object> aggregate =
                (Map<String, Object>)
                        aggregateObject;

        String function =
                getRequiredString(
                        aggregate,
                        "function")
                        .toUpperCase(Locale.ROOT);

        validateAggregateFunction(function);

        Object fieldObject =
                aggregate.get("field");

        String field =
                fieldObject == null
                        ? "*"
                        : fieldObject.toString();

        StringBuilder expression =
                new StringBuilder();

        expression
                .append(function)
                .append("(")
                .append(field)
                .append(")");

        /*
         * FILTER
         *
         * MAX(b.value)
         * FILTER (WHERE b.type = 'GCS')
         */
        Object filterObject =
                aggregate.get("filter");

        if (filterObject instanceof Map) {

            String filterSql =
                    handleQueryClause(
                            (Map<String, Object>)
                                    filterObject);

            if (filterSql != null &&
                    !filterSql.isBlank()) {

                expression
                        .append(" FILTER (WHERE ")
                        .append(filterSql)
                        .append(")");
            }
        }

        /*
         * Alias remains on source item.
         */
        Object aliasObject =
                selectItem.get("alias");

        if (aliasObject instanceof String &&
                !((String) aliasObject).isBlank()) {

            expression
                    .append(" AS ")
                    .append(
                            formatAlias(
                                    (String) aliasObject));
        }

        return expression.toString();
    }


    private void validateAggregateFunction(
            String function) {

        Set<String> allowed =
                new HashSet<>(
                        Arrays.asList(
                                "MAX",
                                "MIN",
                                "SUM",
                                "AVG",
                                "COUNT",
                                "BOOL_OR",
                                "BOOL_AND"));

        if (!allowed.contains(function)) {

            throw new IllegalArgumentException(
                    "Unsupported aggregate function: "
                            + function);
        }
    }


    /*
     * =========================================================
     * FROM EXTRACTION
     *
     * Matches screenshot:
     *
     * String:
     *   "from":"account a"
     *
     * Object:
     *   "from":{
     *      "table":"account",
     *      "alias":"a"
     *   }
     * =========================================================
     */

    private String extractFromClause(
            Map<String, Object> queryRequest) {

        Object fromObject =
                queryRequest.get("from");

        if (fromObject instanceof String) {

            return ((String) fromObject).trim();
        }

        if (fromObject instanceof Map) {

            Map<String, Object> fromMap =
                    (Map<String, Object>)
                            fromObject;

            String table =
                    (String) fromMap.get("table");

            String alias =
                    (String) fromMap.get("alias");

            if (table == null ||
                    table.isBlank()) {

                return null;
            }

            return alias != null &&
                    !alias.isBlank()
                    ? table + " " + alias
                    : table;
        }

        return null;
    }


    /*
     * =========================================================
     * FROM + JOINS
     * =========================================================
     */

    private String handleFromClause(
            Map<String, Object> queryRequest,
            Set<String> cteNames) {

        String fromClause =
                extractFromClause(
                        queryRequest);

        if (fromClause == null) {

            throw new IllegalArgumentException(
                    "Missing from clause in request.");
        }

        String resolvedFrom =
                resolveTableOrCte(
                        fromClause,
                        cteNames);

        Object joinsObject =
                queryRequest.get("joins");

        if (!(joinsObject instanceof List)) {

            return resolvedFrom;
        }

        List<Map<String, Object>> joins =
                (List<Map<String, Object>>)
                        joinsObject;

        StringBuilder joinClause =
                new StringBuilder(
                        resolvedFrom);

        for (Map<String, Object> join :
                joins) {

            if (join == null) {
                continue;
            }

            String joinType =
                    join.get("type") == null
                            ? "INNER"
                            : join.get("type")
                            .toString();

            String joinTableName =
                    getRequiredString(
                            join,
                            "table");

            String joinAlias =
                    join.get("alias") == null
                            ? null
                            : join.get("alias")
                            .toString();

            String type =
                    normalizeJoinType(
                            joinType);

            String joinTable =
                    resolveTableOrCte(
                            joinTableName,
                            cteNames);

            joinClause
                    .append(" ")
                    .append(type)
                    .append(" JOIN ")
                    .append(joinTable);

            if (joinAlias != null &&
                    !joinAlias.isBlank()) {

                joinClause
                        .append(" ")
                        .append(joinAlias);
            }

            if (!"CROSS".equals(type)) {

                String on =
                        getRequiredString(
                                join,
                                "on");

                joinClause
                        .append(" ON ")
                        .append(on);
            }
        }

        return joinClause.toString();
    }


    /*
     * =========================================================
     * PHYSICAL TABLE VS CTE
     * =========================================================
     */

    private String resolveTableOrCte(
            String tableExpression,
            Set<String> cteNames) {

        String expression =
                tableExpression.trim();

        String[] parts =
                expression.split("\\s+");

        String table =
                parts[0];

        String alias =
                parts.length > 1
                        ? parts[1]
                        : null;

        boolean cte =
                cteNames != null &&
                cteNames.contains(
                        table.toLowerCase(
                                Locale.ROOT));

        String resolved;

        if (cte) {

            /*
             * CTE:
             *
             * FROM base b
             *
             * NOT schema.base
             */
            resolved = table;

        } else if (table.contains(".")) {

            resolved = table;

        } else {

            /*
             * Existing behavior.
             */
            resolved =
                    tableConfig.getSchema()
                            + "."
                            + table;
        }

        if (alias != null &&
                !alias.isBlank()) {

            resolved =
                    resolved
                            + " "
                            + alias;
        }

        return resolved;
    }


    /*
     * =========================================================
     * JOIN TYPE
     * =========================================================
     */

    private String normalizeJoinType(
            String joinType) {

        if (joinType == null ||
                joinType.isBlank()) {

            return "INNER";
        }

        String type =
                joinType.trim()
                        .toUpperCase(
                                Locale.ROOT);

        switch (type) {

            case "JOIN":
            case "INNER":
            case "INNER JOIN":
                return "INNER";

            case "LEFT":
            case "LEFT JOIN":
                return "LEFT";

            case "LEFT OUTER":
            case "LEFT OUTER JOIN":
                return "LEFT OUTER";

            case "RIGHT":
            case "RIGHT JOIN":
                return "RIGHT";

            case "RIGHT OUTER":
            case "RIGHT OUTER JOIN":
                return "RIGHT OUTER";

            case "FULL":
            case "FULL JOIN":
                return "FULL";

            case "FULL OUTER":
            case "FULL OUTER JOIN":
                return "FULL OUTER";

            case "CROSS":
            case "CROSS JOIN":
                return "CROSS";

            default:

                throw new IllegalArgumentException(
                        "Unsupported join type: "
                                + joinType);
        }
    }


    /*
     * =========================================================
     * EXISTING QUERY.BOOL
     * =========================================================
     */

    private String handleQueryClause(
            Map<String, Object> queryClause) {

        if (queryClause == null) {
            return "";
        }

        Object boolObject =
                queryClause.get("bool");

        if (!(boolObject instanceof Map)) {
            return "";
        }

        Map<String, Object> boolClause =
                (Map<String, Object>)
                        boolObject;

        StringBuilder whereString =
                new StringBuilder();

        if (boolClause.containsKey("must")) {

            Object mustObject =
                    boolClause.get("must");

            if (mustObject instanceof List) {

                whereString.append(
                        handleMustClause(
                                (List<Map<String, Object>>)
                                        mustObject));
            }
        }

        return whereString.toString();
    }


    /*
     * =========================================================
     * MUST
     * =========================================================
     */

    private String handleMustClause(
            List<Map<String, Object>> mustClauses) {

        StringBuilder mustString =
                new StringBuilder();

        if (mustClauses == null) {
            return "";
        }

        for (Map<String, Object> condition :
                mustClauses) {

            String conditionSql =
                    translateCondition(
                            condition);

            if (conditionSql == null ||
                    conditionSql.isBlank()) {

                continue;
            }

            if (mustString.length() > 0) {
                mustString.append(" AND ");
            }

            mustString.append(
                    conditionSql);
        }

        return mustString.toString();
    }


    /*
     * =========================================================
     * CONDITION
     * =========================================================
     */

    private String translateCondition(
            Map<String, Object> condition) {

        if (condition == null) {
            return "";
        }

        /*
         * MATCH
         */
        if (condition.containsKey("match")) {

            Map<String, Object> matchClause =
                    (Map<String, Object>)
                            condition.get("match");

            List<String> predicates =
                    new ArrayList<>();

            for (Map.Entry<String, Object> entry :
                    matchClause.entrySet()) {

                String column =
                        convertJsonPathToSql(
                                entry.getKey());

                Object value =
                        entry.getValue();

                if (value == null) {

                    predicates.add(
                            column + " IS NULL");

                } else {

                    predicates.add(
                            column
                                    + " = "
                                    + toSqlValue(value));
                }
            }

            return String.join(
                    " AND ",
                    predicates);
        }

        /*
         * RANGE
         */
        if (condition.containsKey("range")) {

            Map<String, Object> rangeClause =
                    (Map<String, Object>)
                            condition.get("range");

            List<String> predicates =
                    new ArrayList<>();

            for (Map.Entry<String, Object> entry :
                    rangeClause.entrySet()) {

                String column =
                        convertJsonPathToSql(
                                entry.getKey());

                Map<String, Object> values =
                        (Map<String, Object>)
                                entry.getValue();

                for (Map.Entry<String, Object> range :
                        values.entrySet()) {

                    predicates.add(
                            column
                                    + " "
                                    + convertRangeOperator(
                                    range.getKey())
                                    + " "
                                    + toSqlValue(
                                    range.getValue()));
                }
            }

            return String.join(
                    " AND ",
                    predicates);
        }

        /*
         * BETWEEN
         */
        if (condition.containsKey("between")) {

            Map<String, Object> between =
                    (Map<String, Object>)
                            condition.get("between");

            String field =
                    convertJsonPathToSql(
                            getRequiredString(
                                    between,
                                    "field"));

            return field
                    + " BETWEEN "
                    + toSqlValue(
                    between.get("from"))
                    + " AND "
                    + toSqlValue(
                    between.get("to"));
        }

        /*
         * IN
         */
        if (condition.containsKey("in")) {

            Map<String, Object> in =
                    (Map<String, Object>)
                            condition.get("in");

            String field =
                    convertJsonPathToSql(
                            getRequiredString(
                                    in,
                                    "field"));

            Object valuesObject =
                    in.get("values");

            if (!(valuesObject instanceof List)) {

                throw new IllegalArgumentException(
                        "IN values must be a list");
            }

            List<?> values =
                    (List<?>) valuesObject;

            String valuesSql =
                    values.stream()
                            .map(this::toSqlValue)
                            .collect(
                                    Collectors.joining(", "));

            return field
                    + " IN ("
                    + valuesSql
                    + ")";
        }

        /*
         * IS NULL
         */
        if (condition.containsKey("isNull")) {

            String field =
                    condition.get("isNull")
                            .toString();

            return convertJsonPathToSql(field)
                    + " IS NULL";
        }

        /*
         * IS NOT NULL
         */
        if (condition.containsKey("isNotNull")) {

            String field =
                    condition.get("isNotNull")
                            .toString();

            return convertJsonPathToSql(field)
                    + " IS NOT NULL";
        }

        return "";
    }


    /*
     * =========================================================
     * GROUP BY
     * =========================================================
     */

    private void appendGroupBy(
            StringBuilder sql,
            Map<String, Object> request) {

        Object object =
                request.get("groupBy");

        if (!(object instanceof List)) {
            return;
        }

        List<?> fields =
                (List<?>) object;

        if (fields.isEmpty()) {
            return;
        }

        sql.append(" GROUP BY ");

        sql.append(
                fields.stream()
                        .map(Object::toString)
                        .collect(
                                Collectors.joining(", ")));
    }


    /*
     * =========================================================
     * HAVING
     * =========================================================
     */

    private void appendHaving(
            StringBuilder sql,
            Map<String, Object> request) {

        Object object =
                request.get("having");

        if (!(object instanceof Map)) {
            return;
        }

        String having =
                handleQueryClause(
                        (Map<String, Object>)
                                object);

        if (having != null &&
                !having.isBlank()) {

            sql
                    .append(" HAVING ")
                    .append(having);
        }
    }


    /*
     * =========================================================
     * ORDER BY
     * =========================================================
     */

    private void appendOrderBy(
            StringBuilder sql,
            Map<String, Object> request) {

        Object object =
                request.get("orderBy");

        if (!(object instanceof List)) {
            return;
        }

        List<?> orders =
                (List<?>) object;

        List<String> values =
                new ArrayList<>();

        for (Object orderObject : orders) {

            if (orderObject instanceof String) {

                values.add(
                        orderObject.toString());

                continue;
            }

            if (!(orderObject instanceof Map)) {
                continue;
            }

            Map<String, Object> order =
                    (Map<String, Object>)
                            orderObject;

            String field =
                    getRequiredString(
                            order,
                            "field");

            String direction =
                    order.get("direction") == null
                            ? "ASC"
                            : order.get("direction")
                            .toString()
                            .toUpperCase(Locale.ROOT);

            if (!"ASC".equals(direction) &&
                    !"DESC".equals(direction)) {

                throw new IllegalArgumentException(
                        "Invalid ORDER BY direction: "
                                + direction);
            }

            values.add(
                    field
                            + " "
                            + direction);
        }

        if (!values.isEmpty()) {

            sql
                    .append(" ORDER BY ")
                    .append(
                            String.join(
                                    ", ",
                                    values));
        }
    }


    /*
     * =========================================================
     * HELPERS
     * =========================================================
     */

    private String getRequiredString(
            Map<String, Object> map,
            String key) {

        Object value =
                map.get(key);

        if (!(value instanceof String) ||
                ((String) value).isBlank()) {

            throw new IllegalArgumentException(
                    "Missing " + key);
        }

        return ((String) value).trim();
    }


    private void validateIdentifier(
            String value) {

        if (value == null ||
                !value.matches(
                        "[A-Za-z_][A-Za-z0-9_]*")) {

            throw new IllegalArgumentException(
                    "Invalid identifier: "
                            + value);
        }
    }


    private String formatAlias(
            String alias) {

        String value =
                alias.trim();

        /*
         * Normal aliases from your screenshot:
         *
         * cashAccountNumber
         * securitiesAccountNumber
         * sanction_flag
         */
        if (value.matches(
                "[A-Za-z_][A-Za-z0-9_]*")) {

            return value;
        }

        /*
         * Allows:
         *
         * "Cash Account Number"
         */
        return "\""
                + value.replace(
                "\"",
                "\"\"")
                + "\"";
    }


    private int parseLimit(
            Object value) {

        int limit =
                Integer.parseInt(
                        value.toString());

        if (limit < 0) {

            throw new IllegalArgumentException(
                    "Invalid limit: "
                            + limit);
        }

        return limit;
    }


    private int parseOffset(
            Object value) {

        int offset =
                Integer.parseInt(
                        value.toString());

        if (offset < 0) {

            throw new IllegalArgumentException(
                    "Invalid offset: "
                            + offset);
        }

        return offset;
    }


    private String convertRangeOperator(
            String operator) {

        switch (operator) {

            case "gt":
                return ">";

            case "gte":
                return ">=";

            case "lt":
                return "<";

            case "lte":
                return "<=";

            default:
                throw new IllegalArgumentException(
                        "Invalid range operator: "
                                + operator);
        }
    }


    /*
     * =========================================================
     * VALUE
     * =========================================================
     */

    private String toSqlValue(
            Object value) {

        if (value == null) {
            return "NULL";
        }

        if (value instanceof Number ||
                value instanceof Boolean) {

            return value.toString();
        }

        return "'"
                + value.toString()
                .replace(
                        "'",
                        "''")
                + "'";
    }


    /*
     * =========================================================
     * JSON PATH
     *
     * IMPORTANT:
     *
     * If your current implementation already handles:
     *
     * alias.field
     * alias.json->field
     * alias.json->>field
     *
     * keep that implementation unchanged.
     * =========================================================
     */

    private String convertJsonPathToSql(
            String column) {

        if (column == null ||
                column.isBlank()) {

            throw new IllegalArgumentException(
                    "Column cannot be empty");
        }

        return column;
    }
